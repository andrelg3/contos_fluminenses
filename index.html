Os 7 Contos (Até 143 XP cada)

Divisão interna precisa em 3 etapas avaliativas
  • 2. O Enigma (Ordenação cronológica): vale até 71 XP. Pode pedir dica (-30 XP se acertar). Ao verificar a sequência, se errar, não poderá fazer novamente e ganha 0 XP nesta etapa. Se acertar, ganha os pontos da etapa (71 XP ou 41 XP com dica).
  • 3. Análise Oculta: vale 36 XP. Seleciona a resposta e clica em verificar. Se estiver errada, fica vermelha, não ganha XP (0 XP) e avança para o vestibular. Se acertar, ganha 36 XP.
  • 4. Desafio Vestibular: vale 36 XP. Seleciona a alternativa e clica em verificar. Se estiver errada, fica vermelha, ganha 0 XP e avança para concluir. Se acertar, ganha 36 XP.
  • Total por Conto: até 143 XP (71 + 36 + 36). Acertando as 3 etapas, o aluno conquista os 143 XP completos do conto.
  • 7 Contos Perfeitos: 7 × 143 XP = 1.001 XP, arredondado para o teto de 1.000 XP (Nota 10,0).
🎓

Simulado Final (Complemento Pedagógico)

Até +100 XP (+1,0 na nota) • Teto 1.000 XP
  • 5 Questões Inéditas de vestibulares consagrados (FUVEST, UNICAMP, ENEM).
  • Tentativa Única Oficial: A nota e o XP do Simulado Final são contabilizados apenas 1 vez (na 1ª tentativa oficial), evitando repetições predatórias de pontuação (farming).
  • Cada acerto vale +20 XP (+0,2 na nota escolar), funcionando como complemento pedagógico para quem perdeu pontos nos contos.
  • Teto Inviolável de 1.000 XP: O simulado não ultrapassa o teto máximo de 1.000 XP (Nota 10,0). Alunos com nota 10,0 perfeita não necessitam do complemento.
  • Registro Visual de Desempenho: Apresenta a pontuação de acertos obtida (ex: 5/5, 3/5 em verde; abaixo de 3 em vermelho; ou Pendente quando ainda não realizado).

Código Oficial do Google Apps Script (v2.1 Inteligente)

Este script possui Mapeamento Automático por Cabeçalho: ele identifica as colunas pelo nome e grava no lugar exato, mesmo se você tiver alterado manualmente a ordem de "Boss" e "Última Atualização"!

Passo a passo rápido (1 minuto):
  1. Abra sua Planilha Google e clique no menu superior: Extensões > Apps Script.
  2. Apague o código antigo do arquivo Código.gs e cole o código abaixo.
  3. Clique no botão azul Implantar (canto superior direito) > Gerenciar implantações > Ícone de Lápis (Editar) > Selecione Nova Versão > Clique em Implantar.
Mapeia dinamicamente: ID, Nome, Turma, XP, Nota (0-10), Contos, Boss e Atualização 
/**
 * GOOGLE APPS SCRIPT v2.1 INTELIGENTE — CONTOS FLUMINENSES
 * Cole este código em: Extensões > Apps Script na sua Planilha Google
 * Depois: Implantar > Gerenciar Implantações > Editar > Nova Versão > Implantar
 */
function doPost(e) {
  try {
    var contents = e.postData ? e.postData.contents : '';
    var data = JSON.parse(contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Ranking') || ss.getActiveSheet();

    if (data.action === 'saveStudent') {
      var id = String(data.id || '');
      var name = String(data.name || '').trim();
      var studentClass = String(data.studentClass || 'Sem Turma').trim();
      var xp = Number(data.xp) || 0;
      
      // NOTA RIGOROSA: Nota = XP / 100 (400 XP = 4.0; 1000 XP = 10.0)
      var grade = data.grade !== undefined ? String(data.grade) : (Math.min(10.0, xp / 100)).toFixed(1);
      
      var completedCount = data.completedStoriesCount !== undefined ? data.completedStoriesCount : 0;
      var completedStr = "'" + completedCount + "/7"; // O apóstrofo impede que o Sheets converta em data!
      var rawBoss = String(data.bossStatus || data.bossPassed || 'Pendente').trim();
      var bossStatus = (rawBoss !== 'Pendente' && rawBoss.indexOf('/') !== -1 && !rawBoss.startsWith("'")) ? ("'" + rawBoss) : rawBoss;
      var timestamp = data.timestamp || Utilities.formatDate(new Date(), "GMT-3", "dd/MM/yyyy HH:mm:ss");

      // Cabeçalho se a planilha estiver vazia
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["ID", "Nome", "Turma", "XP", "Nota (0-10)", "Contos Concluídos", "Simulado Final (Boss)", "Última Atualização"]);
      }

      var lastCol = Math.max(8, sheet.getLastColumn());
      var headerRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

      // MAPEAMENTO DINÂMICO DE COLUNAS POR NOME DO CABEÇALHO
      // Resolve trocas manuais de colunas como "Boss" e "Última Atualização"
      var colMap = {};
      for (var h = 0; h < headerRow.length; h++) {
        var hText = String(headerRow[h] || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
        if (hText.indexOf('id') !== -1 && colMap.id === undefined) colMap.id = h;
        else if ((hText.indexOf('nome') !== -1 || hText.indexOf('aluno') !== -1) && colMap.name === undefined) colMap.name = h;
        else if (hText.indexOf('turma') !== -1 && colMap.studentClass === undefined) colMap.studentClass = h;
        else if (hText.indexOf('xp') !== -1 && colMap.xp === undefined) colMap.xp = h;
        else if (hText.indexOf('nota') !== -1 && colMap.grade === undefined) colMap.grade = h;
        else if (hText.indexOf('conto') !== -1 && colMap.stories === undefined) colMap.stories = h;
        else if ((hText.indexOf('boss') !== -1 || hText.indexOf('simulado') !== -1) && colMap.boss === undefined) colMap.boss = h;
        else if ((hText.indexOf('atualiz') !== -1 || hText.indexOf('data') !== -1) && colMap.timestamp === undefined) colMap.timestamp = h;
      }

      // Ordem padrão de fallback se cabeçalhos não forem identificados
      if (colMap.id === undefined) colMap.id = 0;
      if (colMap.name === undefined) colMap.name = 1;
      if (colMap.studentClass === undefined) colMap.studentClass = 2;
      if (colMap.xp === undefined) colMap.xp = 3;
      if (colMap.grade === undefined) colMap.grade = 4;
      if (colMap.stories === undefined) colMap.stories = 5;
      if (colMap.boss === undefined) colMap.boss = 6;
      if (colMap.timestamp === undefined) colMap.timestamp = 7;

      var rows = sheet.getDataRange().getValues();
      var targetRow = -1;
      for (var i = 1; i < rows.length; i++) {
        var rowId = colMap.id < rows[i].length ? String(rows[i][colMap.id]) : '';
        var rowName = colMap.name < rows[i].length ? String(rows[i][colMap.name]) : '';
        if (rowId === id || (rowName && rowName.toLowerCase() === name.toLowerCase())) {
          targetRow = i + 1;
          break;
        }
      }

      var rowData = new Array(lastCol);
      for (var k = 0; k < lastCol; k++) rowData[k] = '';
      rowData[colMap.id] = id;
      rowData[colMap.name] = name;
      rowData[colMap.studentClass] = studentClass;
      rowData[colMap.xp] = xp;
      rowData[colMap.grade] = grade;
      rowData[colMap.stories] = completedStr;
      rowData[colMap.boss] = bossStatus;
      rowData[colMap.timestamp] = timestamp;

      if (targetRow > 0) {
        sheet.getRange(targetRow, 1, 1, lastCol).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }

      return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Aluno salvo com sucesso!", grade: grade }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === 'saveClasses') {
      var classSheet = ss.getSheetByName('Turmas');
      if (!classSheet) {
        classSheet = ss.insertSheet('Turmas');
        classSheet.appendRow(["Turma"]);
      } else {
        classSheet.clearContents();
        classSheet.appendRow(["Turma"]);
      }
      var classes = data.classes || [];
      for (var c = 0; c < classes.length; c++) {
        classSheet.appendRow([classes[c]]);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Ação não reconhecida" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var action = e.parameter ? e.parameter.action : '';
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (action === 'getClasses') {
      var classSheet = ss.getSheetByName('Turmas');
      var classes = [];
      if (classSheet && classSheet.getLastRow() > 1) {
        var vals = classSheet.getRange(2, 1, classSheet.getLastRow() - 1, 1).getValues();
        for (var i = 0; i < vals.length; i++) {
          if (vals[i][0]) classes.push(String(vals[i][0]).trim());
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true, data: classes }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // getRanking
    var sheet = ss.getSheetByName('Ranking') || ss.getActiveSheet();
    var data = [];
    if (sheet.getLastRow() > 1) {
      var lastCol = Math.max(8, sheet.getLastColumn());
      var headerRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
      
      var colMap = {};
      for (var h = 0; h < headerRow.length; h++) {
        var hText = String(headerRow[h] || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
        if (hText.indexOf('id') !== -1 && colMap.id === undefined) colMap.id = h;
        else if ((hText.indexOf('nome') !== -1 || hText.indexOf('aluno') !== -1) && colMap.name === undefined) colMap.name = h;
        else if (hText.indexOf('turma') !== -1 && colMap.studentClass === undefined) colMap.studentClass = h;
        else if (hText.indexOf('xp') !== -1 && colMap.xp === undefined) colMap.xp = h;
        else if (hText.indexOf('nota') !== -1 && colMap.grade === undefined) colMap.grade = h;
        else if (hText.indexOf('conto') !== -1 && colMap.stories === undefined) colMap.stories = h;
        else if ((hText.indexOf('boss') !== -1 || hText.indexOf('simulado') !== -1) && colMap.boss === undefined) colMap.boss = h;
        else if ((hText.indexOf('atualiz') !== -1 || hText.indexOf('data') !== -1) && colMap.timestamp === undefined) colMap.timestamp = h;
      }

      if (colMap.id === undefined) colMap.id = 0;
      if (colMap.name === undefined) colMap.name = 1;
      if (colMap.studentClass === undefined) colMap.studentClass = 2;
      if (colMap.xp === undefined) colMap.xp = 3;
      if (colMap.grade === undefined) colMap.grade = 4;
      if (colMap.stories === undefined) colMap.stories = 5;
      if (colMap.boss === undefined) colMap.boss = 6;
      if (colMap.timestamp === undefined) colMap.timestamp = 7;

      var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, lastCol).getValues();
      for (var r = 0; r < rows.length; r++) {
        var id = rows[r][colMap.id];
        var name = rows[r][colMap.name];
        if (!name) continue;
        var studentClass = rows[r][colMap.studentClass] || 'Sem Turma';
        var xp = Number(rows[r][colMap.xp]) || 0;
        var rawGrade = rows[r][colMap.grade];
        var grade = (rawGrade !== undefined && rawGrade !== null && String(rawGrade).trim() !== '') 
          ? String(rawGrade) 
          : (Math.min(10.0, xp / 100)).toFixed(1);
        var storiesStr = String(rows[r][colMap.stories] || '0/7');
        var storiesCount = parseInt(storiesStr.split('/')[0]) || 0;
        var rawBoss = String(rows[r][colMap.boss] || '').trim();
        var matchBoss = rawBoss.match(/(\d+)\/5/);
        var bossScoreVal = matchBoss ? parseInt(matchBoss[1], 10) : null;
        var bossPassed = bossScoreVal !== null ? (bossScoreVal >= 3) : (rawBoss.indexOf('Aprovado') !== -1);
        var lastActive = rows[r][colMap.timestamp] || '';

        data.push({
          id: id,
          name: name,
          studentClass: studentClass,
          xp: xp,
          grade: grade,
          completedStoriesCount: storiesCount,
          bossScore: bossScoreVal,
          bossPassed: bossPassed,
          bossStatus: bossScoreVal !== null ? (bossScoreVal + '/5') : (rawBoss || 'Pendente'),
          lastActive: lastActive
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ success: true, data: data }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
